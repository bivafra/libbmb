134template <InputIterator Iter>
135auto distance(Iter first, Iter last) -> IteratorTraits<Iter>::difference_type {
136 using diff_t = IteratorTraits<Iter>::difference_type;
137 using iter_category = IteratorTraits<Iter>::iterator_category;
138
139 // If can - do it for O(1), otherwise iterate through all range
140 if constexpr (is_base_of_v<random_access_iter_tag, iter_category>) {
141 return last - first;
142 }
143
144 diff_t n = 0;
145 while (first != last) {
146 ++first;
147 ++n;
148 }
149 return n;
150}

167template <InputIterator Iter, typename Distance>
168void advance(Iter& it, Distance dist) {
169 using diff_t = IteratorTraits<Iter>::difference_type;
170 using iter_category = IteratorTraits<Iter>::iterator_category;
171
172 auto n = static_cast<diff_t>(dist);
173
174 // Make operations according to the iterator category
175 if constexpr (is_base_of_v<random_access_iter_tag, iter_category>)
176 it += n;
